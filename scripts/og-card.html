DECLARE start_date DATE DEFAULT '2025-01-01';
-- weekly revenue per country
WITH orders AS (
  SELECT user_id, amount, order_date
  FROM shop.orders
  WHERE order_date >= start_date
),
by_week AS (
  SELECT u.country, DATE_TRUNC(o.order_date, WEEK) AS wk,
         SUM(o.amount) AS revenue
  FROM orders o JOIN shop.users u USING (user_id)
  GROUP BY 1, 2
)
